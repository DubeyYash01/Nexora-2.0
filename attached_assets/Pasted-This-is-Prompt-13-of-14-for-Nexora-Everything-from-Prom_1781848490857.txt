export const analyzeSchema = z.object({

    .optional().nullable(),

export const chatSchema = z.object({
  message: z.string()
    .min(1, 'Message required')
    .max(1000, 'Message too long'),
  projectId: z.string().uuid()
    .optional().nullable(),
  userId: z.string().uuid()
    .optional().nullable(),
  projectContext: z.string()
    .optional(),
  conversationHistory: z.array(
    z.object({
      role: z.enum(['user', 'assistant']),
      content: z.string()
    })
  ).optional().default([])
})

export function validate(schema: any) {
  return (
    req: Request, 
    res: Response, 
    next: NextFunction
  ) => {
    try {
      req.body = schema.parse(req.body)
      next()
    } catch (err: any) {
      res.status(400).json({
        error: 'Invalid request',
        details: err.errors?.map(
          (e: any) => e.message
        )
      })
    }
  }
}

import NodeCache from 'node-cache'

// Cache with 10 minute TTL
const cache = new NodeCache({
  stdTTL: 600,
  checkperiod: 120,
  useClones: false
})

export function cacheMiddleware(
  ttl: number = 600
) {
  return (
    req: any,
    res: any,
    next: any
  ) => {
    const key = `${req.method}:
      ${req.originalUrl}`
    const cached = cache.get(key)
    
    if (cached) {
      return res.json(cached)
    }
    
    const originalJson = res.json
      .bind(res)
    
    res.json = (data: any) => {
      if (res.statusCode === 200) {
        cache.set(key, data, ttl)
      }
      return originalJson(data)
    }
    
    next()
  }
}

export { cache }

Apply caching to appropriate routes:

// Cache blueprints for 10 minutes

  cacheMiddleware(600),
  async (req, res) => { ... })

router.get('/recommended',
  cacheMiddleware(600),
  async (req, res) => { ... })

// Cache blueprint list for 5 minutes

  cacheMiddleware(300),
  async (req, res) => { ... })

// Do NOT cache:
// - AI routes (always fresh)
// - User-specific data
// - POST routes

    }

    health.services.database?.status === 
    'error' ? 503 : 200

export function serveProductionBuild(
  app: Express
): void {

        error: 'API route not found'
      })

    res.sendFile(
      path.join(
        clientBuildPath, 
        'index.html'
      )
    )
  })

import { useState, useRef, 
         useEffect } from 'react'

interface SerialLine {
  id: string
  text: string
  timestamp: string
  isError: boolean
}

export default function SerialMonitor() {
  const [connected, setConnected] = 
    useState(false)
  const [lines, setLines] = 
    useState<SerialLine[]>([])
  const [baudRate, setBaudRate] = 
    useState(115200)
  const [input, setInput] = 
    useState('')
  const portRef = useRef<any>(null)
  const writerRef = useRef<any>(null)
  const scrollRef = useRef<HTMLDivElement>
    (null)

  useEffect(() => {
    if (scrollRef.current) {
      scrollRef.current.scrollTop =
        scrollRef.current.scrollHeight
    }
  }, [lines])

  const addLine = (text: string) => {
    const isError = 
      text.toLowerCase()
        .includes('error') ||
      text.toLowerCase()
        .includes('failed')
    
    setLines(prev => [...prev, {
      id: Date.now().toString(),
      text: text.trim(),
      timestamp: new Date()
        .toLocaleTimeString(),
      isError
    }].slice(-500)) // Keep last 500 lines
  }

  const connect = async () => {
    if (!('serial' in navigator)) {
      alert(
        'Serial Monitor requires Chrome. 
         Please open Nexora in Chrome.'
      )
      return
    }

    try {
      const port = await 
        (navigator as any).serial
          .requestPort()
      
      await port.open({ baudRate })
      portRef.current = port
      
      if (port.writable) {
        writerRef.current = 
          port.writable.getWriter()
      }
      
      setConnected(true)
      addLine(
        '--- Connected to device ---'
      )
      
      // Start reading
      const reader = 
        port.readable.getReader()
      const decoder = new TextDecoder()
      let buffer = ''
      
      const readLoop = async () => {
        try {
          while (true) {
            const { value, done } = 
              await reader.read()
            if (done) break
            
            buffer += decoder.decode(
              value, { stream: true }
            )
            
            const lineBreak = 
              buffer.lastIndexOf('\n')
            if (lineBreak >= 0) {
              const complete = 
                buffer.substring(
                  0, lineBreak
                )
              buffer = buffer.substring(
                lineBreak + 1
              )
              
              complete.split('\n')
                .filter(l => l.trim())
                .forEach(addLine)
            }
          }
        } catch (err) {
          addLine(
            '--- Device disconnected ---'
          )
          setConnected(false)
        } finally {
          reader.releaseLock()
        }
      }
      
      readLoop()
      
    } catch (err: any) {
      if (err.name !== 'NotFoundError') {
        addLine(`Error: ${err.message}`)
      }
    }
  }

  const disconnect = async () => {
    try {
      if (writerRef.current) {
        writerRef.current.releaseLock()
      }
      if (portRef.current) {
        await portRef.current.close()
      }
    } catch (err) {
      // Ignore close errors
    }
    setConnected(false)
    addLine('--- Disconnected ---')
  }

  const sendCommand = async () => {
    if (!writerRef.current || 
        !input.trim()) return
    
    try {
      const encoder = new TextEncoder()
      await writerRef.current.write(
        encoder.encode(input + '\n')
      )
      addLine(`> ${input}`)
      setInput('')
    } catch (err: any) {
      addLine(`Send error: ${err.message}`)
    }
  }

  const clearOutput = () => setLines([])
  
  const saveLog = () => {
    const content = lines
      .map(l => 
        `[${l.timestamp}] ${l.text}`
      )
      .join('\n')
    const blob = new Blob(
      [content], 
      { type: 'text/plain' }
    )
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = 'nexora-serial-log.txt'
    a.click()
    URL.revokeObjectURL(url)
  }

  return (
    <div className="flex flex-col h-full 
      bg-[#0A0A0F]">
      
      {/* Header */}
      <div className="flex items-center 
        justify-between px-4 py-2 
        border-b border-[#2A2A3E]">
        
        <div className="flex items-center 
          gap-3">
          <div className={`w-2 h-2 
            rounded-full ${connected ? 
              'bg-[#00C896] animate-pulse' : 
              'bg-[#5A5A7A]'
            }`} 
          />
          <span className="text-sm 
            font-medium text-[#F0F0FF]">
            {connected ? 
              'Device Connected' : 
              'No Device'
            }
          </span>
          
          {!connected && (
            <select
              value={baudRate}
              onChange={e => 
                setBaudRate(
                  Number(e.target.value)
                )
              }
              className="text-xs 
                bg-[#12121A] 
                border border-[#2A2A3E]
                text-[#9090B0] 
                rounded px-2 py-1"
            >
              <option value={9600}>
                9600
              </option>
              <option value={57600}>
                57600
              </option>
              <option value={115200}>
                115200
              </option>
            </select>
          )}
        </div>
        
        <div className="flex items-center 
          gap-2">
          <button
            onClick={clearOutput}
            className="text-xs 
              text-[#5A5A7A] 
              hover:text-[#9090B0] 
              px-2 py-1"
          >
            Clear
          </button>
          <button
            onClick={saveLog}
            disabled={lines.length === 0}
            className="text-xs 
              text-[#5A5A7A] 
              hover:text-[#9090B0] 
              px-2 py-1 
              disabled:opacity-30"
          >
            Save Log
          </button>
          <button
            onClick={connected ? 
              disconnect : connect
            }
            className={`text-xs px-3 py-1 
              rounded-lg font-medium
              transition-all duration-200
              ${connected
                ? 'bg-[#FF5A5A] 
                   text-white 
                   hover:bg-red-600'
                : 'bg-[#6C63FF] 
                   text-white 
                   hover:bg-[#5A52E0]'
              }`}
          >
            {connected ? 
              'Disconnect' : 
              'Connect Device'
            }
          </button>
        </div>
      </div>
      
      {/* Chrome warning */}
      {!('serial' in navigator) && (
        <div className="mx-4 mt-2 p-2 
          bg-[rgba(255,184,77,0.08)] 
          border border-[#FFB84D] 
          rounded-lg text-xs 
          text-[#FFB84D]">
          ⚠️ Serial Monitor requires 
          Chrome browser
        </div>
      )}
      
      {/* Output */}
      <div
        ref={scrollRef}
        className="flex-1 overflow-y-auto 
          p-3 font-mono text-xs"
      >
        {lines.length === 0 ? (
          <div className="text-[#3A3A5A] 
            text-center mt-8">
            Connect your device to see 
            serial output here
          </div>
        ) : (
          lines.map(line => (
            <div
              key={line.id}
              className={`mb-0.5 
                leading-5
                ${line.isError
                  ? 'text-[#FF5A5A]'
                  : line.text.startsWith(
                      '---'
                    )
                    ? 'text-[#5A5A7A]'
                    : 'text-[#00C896]'
                }`}
            >
              <span className="text-[#3A3A5A] 
                mr-2">
                [{line.timestamp}]
              </span>
              {line.text}
            </div>
          ))
        )}
      </div>
      
      {/* Input */}
      {connected && (
        <div className="flex gap-2 p-3 
          border-t border-[#2A2A3E]">
          <input
            value={input}
            onChange={e => 
              setInput(e.target.value)
            }
            onKeyDown={e => {
              if (e.key === 'Enter') {
                sendCommand()
              }
            }}
            placeholder="Send command..."
            className="flex-1 bg-[#12121A] 
              border border-[#2A2A3E] 
              rounded-lg px-3 py-1.5 
              text-xs text-[#F0F0FF] 
              font-mono
              focus:border-[#6C63FF] 
              focus:outline-none"
          />
          <button
            onClick={sendCommand}
            className="bg-[#6C63FF] 
              text-white text-xs 
              px-3 py-1.5 rounded-lg
              hover:bg-[#5A52E0]"
          >
            Send
          </button>
        </div>
      )}
    </div>
  )
}
