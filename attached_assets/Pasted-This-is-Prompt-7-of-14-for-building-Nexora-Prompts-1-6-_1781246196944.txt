CREATE TABLE IF NOT EXISTS classes (
  id uuid DEFAULT gen_random_uuid()
     PRIMARY KEY,
  professor_id uuid 
    REFERENCES profiles(id),
  name text NOT NULL,
  subject text,
  college text,
  academic_year text,
  join_code text UNIQUE NOT NULL,
  student_count integer DEFAULT 0,
  is_active boolean DEFAULT true,
  created_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS class_members (
  id uuid DEFAULT gen_random_uuid()
     PRIMARY KEY,
  class_id uuid REFERENCES classes(id)
    ON DELETE CASCADE,
  student_id uuid REFERENCES profiles(id),
  joined_at timestamptz DEFAULT now(),
  UNIQUE(class_id, student_id)
);

CREATE TABLE IF NOT EXISTS assignments (
  id uuid DEFAULT gen_random_uuid()
     PRIMARY KEY,
  class_id uuid REFERENCES classes(id),
  professor_id uuid 
    REFERENCES profiles(id),
  title text NOT NULL,
  description text,
  objectives text[],
  allowed_components jsonb,
  required_phases text[],
  deadline timestamptz,
  max_group_size integer DEFAULT 4,
  allow_any_components boolean 
    DEFAULT true,
  grading_criteria jsonb,
  status text DEFAULT 'active'
    CHECK (status IN (
      'draft','active','closed'
    )),
  submission_count integer DEFAULT 0,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS 
  assignment_submissions (
  id uuid DEFAULT gen_random_uuid()
     PRIMARY KEY,
  assignment_id uuid 
    REFERENCES assignments(id),
  project_id uuid 
    REFERENCES projects(id),
  student_id uuid 
    REFERENCES profiles(id),
  group_members jsonb,
  submitted_at timestamptz,
  status text DEFAULT 'draft'
    CHECK (status IN (
      'draft','submitted','graded'
    )),
  grade text,
  professor_feedback text,
  ai_assistance_log jsonb,
  component_list jsonb,
  build_steps_completed integer,
  total_build_steps integer,
  graded_at timestamptz,
  graded_by uuid 
    REFERENCES profiles(id)
);

ALTER TABLE projects
  ADD COLUMN IF NOT EXISTS
  assignment_id uuid 
  REFERENCES assignments(id) NULLABLE;

ALTER TABLE projects
  ADD COLUMN IF NOT EXISTS
  submitted_for_assignment 
  boolean DEFAULT false;

-- RLS
ALTER TABLE classes 
  ENABLE ROW LEVEL SECURITY;
CREATE POLICY "professors manage classes"
  ON classes FOR ALL
  USING (auth.uid() = professor_id);
CREATE POLICY "students view joined classes"
  ON classes FOR SELECT
  USING (
    EXISTS (
      SELECT 1 FROM class_members
      WHERE class_id = classes.id
      AND student_id = auth.uid()
    )
  );

ALTER TABLE assignments
  ENABLE ROW LEVEL SECURITY;
CREATE POLICY "professors manage assignments"
  ON assignments FOR ALL
  USING (auth.uid() = professor_id);
CREATE POLICY "students view assignments"
  ON assignments FOR SELECT
  USING (
    EXISTS (
      SELECT 1 FROM class_members cm
      JOIN classes c ON c.id = cm.class_id
      WHERE c.id = assignments.class_id
      AND cm.student_id = auth.uid()
    )
  );

ALTER TABLE assignment_submissions
  ENABLE ROW LEVEL SECURITY;
CREATE POLICY "students manage own submissions"
  ON assignment_submissions FOR ALL
  USING (auth.uid() = student_id);
CREATE POLICY "professors view submissions"
  ON assignment_submissions FOR SELECT
  USING (
    EXISTS (
      SELECT 1 FROM assignments a
      WHERE a.id = assignment_id
      AND a.professor_id = auth.uid()
    )
  );
CREATE POLICY "professors update submissions"
  ON assignment_submissions FOR UPDATE
  USING (
    EXISTS (
      SELECT 1 FROM assignments a
      WHERE a.id = assignment_id
      AND a.professor_id = auth.uid()
    )
  );

ALTER TABLE class_members
  ENABLE ROW LEVEL SECURITY;
CREATE POLICY "members manage own membership"
  ON class_members FOR ALL
  USING (auth.uid() = student_id);
CREATE POLICY "professors view class members"
  ON class_members FOR SELECT
  USING (
    EXISTS (
      SELECT 1 FROM classes c
      WHERE c.id = class_id
      AND c.professor_id = auth.uid()
    )
  );

  Body: { studentId, joinCode }
  → Finds class by join code
  → Validates not already member
  → Inserts class_member row
  → Updates student_count
  → Returns { class, assignments }

GET /api/classes/:classId/students
  → Returns all students in class
  → Professor only

POST /api/assignments/create
  Body: { professorId, classId, title,
          description, objectives,
          allowedComponents,
          requiredPhases, deadline,
          maxGroupSize, gradingCriteria,
          status }
  → Creates assignment
  → Returns { assignment }

GET /api/assignments/student/:studentId
  → Returns all assignments for 
    classes student has joined
  → Includes submission status
    for each assignment

GET /api/assignments/:assignmentId/
    submissions
  → Professor only
  → Returns all submissions with 
    student profile data
  → Ordered by submitted_at

GET /api/submissions/:submissionId
  → Returns full submission data
  → Includes project, student profile,
    ai conversation summary

POST /api/submissions/submit
  Body: { projectId, assignmentId,
          studentId, groupMembers,
          videoDemoUrl, studentNote }
  → Creates/updates submission
  → Sets status = 'submitted'
  → Saves ai_assistance_log summary
  → Updates assignment submission_count
  → Returns { submission }

PUT /api/submissions/grade
  Body: { submissionId, professorId,
          grade, feedback, 
          rubricScores }
  → Updates submission with grade
  → Sets status = 'graded'
  → Returns { submission }

GET /api/analytics/:professorId
  Query: classId (optional)
  → Returns aggregated analytics data
  → Completion rates, usage stats
