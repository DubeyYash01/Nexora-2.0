File: client/src/hooks/useCodePush.js

This hook manages all code push logic.

export function useCodePush({
  buildPlan,
  setIdeCode,
  setNewHighlightLines,
  setLibraries,
  setCompletedSteps,
  setCurrentStepIndex,
  projectId,
  userId
}) {

  async function completeStep(stepNumber) {
    
    const step = buildPlan.steps.find(
      s => s.stepNumber === stepNumber
    )
    if (!step) return
    
    const newCode = step.code.content
    const highlightLines = 
      step.code.highlightLines || []
    
    // 1. Update IDE code
    setIdeCode(newCode)
    
    // 2. Trigger highlight animation
    setNewHighlightLines(highlightLines)
    
    // Clear highlights after 2.5 seconds
    setTimeout(() => {
      setNewHighlightLines([])
    }, 2500)
    
    // 3. Add new libraries
    const newLibraries = 
      (step.libraries || [])
      .filter(lib => lib.isNew)
    
    setLibraries(prev => {
      const existing = 
        prev.map(l => l.name)
      const toAdd = newLibraries
        .filter(l => 
          !existing.includes(l.name)
        )
      return [...prev, ...toAdd]
    })
    
    // 4. Mark step complete
    setCompletedSteps(prev => 
      [...prev, stepNumber]
    )
    
    // 5. Move to next step
    setCurrentStepIndex(
      stepNumber  
      // (stepNumber is 1-based,
      //  next step index = stepNumber)
    )
    
    // 6. Save to Supabase
    await saveProgress({
      projectId,
      stepNumber,
      code: newCode,
      libraries: newLibraries
    })
    
    // 7. Show success toast
    // (call toast from outside hook)
    return {
      success: true,
      newCode,
      highlightLines,
      newLibraries
    }
  }
  
  async function saveProgress({
    projectId, stepNumber, 
    code, libraries
  }) {
    try {
      await fetch(
        '/api/projects/complete-step',
        {
          method: 'POST',
          headers: {
            'Content-Type': 
              'application/json'
          },
          body: JSON.stringify({
            projectId,
            stepNumber,
            ideCode: code,
            completedSteps: stepNumber
          })
        }
      )
    } catch (err) {
      console.error(
        'Failed to save progress:', 
        err
      )
    }
  }
  
  return { completeStep }
}
