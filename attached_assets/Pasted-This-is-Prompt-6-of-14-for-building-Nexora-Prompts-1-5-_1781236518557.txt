CREATE TABLE IF NOT EXISTS blueprints (
  id uuid DEFAULT gen_random_uuid() 
     PRIMARY KEY,
  author_id uuid REFERENCES profiles(id),
  title text NOT NULL,
  description text,
  difficulty text CHECK (
    difficulty IN (
      'Beginner','Intermediate','Advanced'
    )
  ),
  category text,
  components jsonb,
  build_plan jsonb,
  ai_analysis jsonb,
  tags text[],
  is_featured boolean DEFAULT false,
  is_public boolean DEFAULT true,
  fork_count integer DEFAULT 0,
  view_count integer DEFAULT 0,
  like_count integer DEFAULT 0,
  source_project_id uuid 
    REFERENCES projects(id) 
    NULLABLE,
  estimated_cost_min integer,
  estimated_cost_max integer,
  estimated_time text,
  platform text,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS blueprint_likes (
  id uuid DEFAULT gen_random_uuid() 
     PRIMARY KEY,
  blueprint_id uuid 
    REFERENCES blueprints(id) 
    ON DELETE CASCADE,
  user_id uuid REFERENCES profiles(id),
  created_at timestamptz DEFAULT now(),
  UNIQUE(blueprint_id, user_id)
);

CREATE TABLE IF NOT EXISTS blueprint_forks (
  id uuid DEFAULT gen_random_uuid() 
     PRIMARY KEY,
  blueprint_id uuid 
    REFERENCES blueprints(id),
  forked_by uuid REFERENCES profiles(id),
  new_project_id uuid 
    REFERENCES projects(id),
  forked_at timestamptz DEFAULT now()
);

ALTER TABLE projects 
  ADD COLUMN IF NOT EXISTS 
  is_public boolean DEFAULT false;

ALTER TABLE projects 
  ADD COLUMN IF NOT EXISTS 
  share_token text UNIQUE;

ALTER TABLE projects 
  ADD COLUMN IF NOT EXISTS 
  blueprint_id uuid 
  REFERENCES blueprints(id) NULLABLE;

ALTER TABLE projects
  ADD COLUMN IF NOT EXISTS
  forked_from uuid 
  REFERENCES blueprints(id) NULLABLE;

-- RLS policies
ALTER TABLE blueprints 
  ENABLE ROW LEVEL SECURITY;

CREATE POLICY "public blueprints readable" 
  ON blueprints FOR SELECT 
  USING (is_public = true);

CREATE POLICY "authors manage blueprints" 
  ON blueprints FOR ALL 
  USING (auth.uid() = author_id);

ALTER TABLE blueprint_likes 
  ENABLE ROW LEVEL SECURITY;

CREATE POLICY "users manage own likes" 
  ON blueprint_likes FOR ALL 
  USING (auth.uid() = user_id);

ALTER TABLE blueprint_forks 
  ENABLE ROW LEVEL SECURITY;

CREATE POLICY "users manage own forks" 
  ON blueprint_forks FOR ALL 
  USING (auth.uid() = forked_by);

FORK A BLUEPRINT FLOW
═══════════════════════════════════════

When user clicks "Fork This Blueprint":

1. Show fork modal:
   
   Heading: "Fork: [Blueprint Title]"
   
   Field 1: Project Name
     Pre-filled: "[Blueprint Title] — 
                  [user's first name]'s version"
     Editable
   
   Field 2: Customize for my components
     Toggle: ON by default
     "Nexora will adapt this blueprint 
      to your component inventory"
   
   Field 3: Skill level confirmation
     Shows detected skill level from profile
     Allow override
   
   "Start Building" button (primary)
   "Cancel" button

2. On confirm:
   
   POST /api/blueprints/fork
   
   Backend:
   a. Creates new project row in projects:
      - title = customized name
      - idea_input = blueprint description
      - ai_analysis = blueprint ai_analysis
      - components = blueprint components
        (with owned flags from user inventory)
      - build_plan = blueprint build_plan
      - status = 'in_progress'
      - forked_from = blueprint id
   
   b. If "customize for my components":
      Call Gemini to adapt:
      
      Prompt:
      "A user is forking this IoT blueprint:
      [blueprint title and description]
      
      Original components: [list]
      User's inventory: [list]
      
      Adapt the component list:
      - Mark components as owned if in inventory
      - Suggest substitutions for missing ones
      - Keep the project functional
      
      Return JSON:
      {
        'adaptedComponents': [...],
        'changes': ['change 1', 'change 2'],
        'totalNewCost': number
      }"
   
   c. Increments fork_count in blueprints
   
   d. Inserts row in blueprint_forks
   
   e. Returns { projectId }
   
   f. Redirects to /workspace/[projectId]
   
   Show adaptation changes summary 
   as first AI assistant message:
   "I've adapted this blueprint for 
    your components. Here's what changed:
    [list of changes]"

3. Toast on redirect:
   "Blueprint forked! Your customized 
    project is ready to build."

═══════════════════════════════════════
BACKEND ROUTES TO ADD
═══════════════════════════════════════

GET /api/blueprints
  Query params: 
    search, difficulty, category, 
    sort, limit, offset
  → Returns filtered blueprints
  → Includes author profile data
  → Includes user's like status
    if authenticated

GET /api/blueprints/:id
  → Returns full blueprint data
  → Increments view_count
  → Includes reviews with user data

GET /api/blueprints/seed
  → Seeds 6 official blueprints 
    if table empty

POST /api/blueprints/fork
  Body: { blueprintId, userId, 
          projectName, adaptToInventory }
  → Creates project from blueprint
  → Calls Gemini if adaptToInventory
  → Increments fork_count
  → Returns { projectId }

POST /api/blueprints/publish
  Body: { projectId, userId, title,
          description, category, tags,
          difficulty, isPublic, 
          showAuthor }
  → Creates blueprint row
  → Links to project
  → Returns { blueprintId, url }

POST /api/blueprints/like
  Body: { blueprintId, userId }
  → Toggles like in blueprint_likes
  → Updates like_count in blueprints
  → Returns { liked: boolean, count }

POST /api/blueprints/review
  Body: { blueprintId, userId, 
          rating, reviewText }
  → Upserts in blueprint_reviews
  → Returns { review }

POST /api/projects/share
  Body: { projectId, userId, 
          makePublic }
  → Generates share_token
