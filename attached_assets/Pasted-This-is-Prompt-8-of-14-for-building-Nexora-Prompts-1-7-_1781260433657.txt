File: client/src/hooks/usePlan.js

export function usePlan() {
  const { profile } = useAuth()
  const plan = profile?.plan || 'free'
  const limits = PLAN_LIMITS[plan]
  
  function canDo(feature) {
    return limits[feature] === true ||
           limits[feature] === -1 ||
           (typeof limits[feature] === 
            'number' && 
            limits[feature] > 0)
  }
  
  function isLimited(feature, current) {
    const limit = limits[feature]
    if (limit === -1) return false
    return current >= limit
  }
  
  return { plan, limits, canDo, isLimited }
}

PAYMENT FLOW
────────────────────────────────────

When user clicks upgrade button:

STEP 1 — Create Order (backend):
POST /api/payments/create-order
Body: { userId, plan, billingCycle }

Backend:
  Calculates amount:
    student_pro monthly: 29900 (paise)
    student_pro semester: 99900
    maker_pro monthly: 49900
  
  Creates Razorpay order:
  razorpay.orders.create({
    amount: amount,
    currency: 'INR',
    receipt: `nexora_${userId}_${Date.now()}`,
    notes: { userId, plan, billingCycle }
  })
  
  Returns { orderId, amount, currency,
            keyId }

STEP 2 — Open Razorpay Checkout 
(frontend):

Load Razorpay script dynamically:
const script = document.createElement('script')
script.src = 
  'https://checkout.razorpay.com/v1/checkout.js'
document.body.appendChild(script)

Open checkout:
const options = {
  key: keyId,
  amount: amount,
  currency: 'INR',
  name: 'Nexora',
  description: `${planName} Plan`,
  order_id: orderId,
  prefill: {
    name: user.full_name,
    email: user.email
  },
  theme: {
    color: '#6C63FF'
  },
  handler: function(response) {
    verifyPayment(response)
  },
  modal: {
    ondismiss: function() {
      setPaymentCancelled(true)
    }
  }
}
const rzp = new window.Razorpay(options)
rzp.open()

STEP 3 — Verify Payment (backend):
POST /api/payments/verify
Body: { 
  razorpay_order_id,
  razorpay_payment_id,
  razorpay_signature,
  userId, plan, billingCycle
}

Backend verification:
  const crypto = require('crypto')
  const hmac = crypto.createHmac(
    'sha256', 
    RAZORPAY_KEY_SECRET
  )
  hmac.update(
    orderId + '|' + paymentId
  )
  const generated = hmac.digest('hex')
  
  if (generated !== signature):
    return { success: false, 
             error: 'Invalid signature' }
  
  If valid:
    Calculate period end:
      monthly: now + 30 days
      semester: now + 180 days
    
    Upsert subscriptions table:
      plan = plan
      billing_cycle = billingCycle
      status = 'active'
      razorpay_payment_id = paymentId
      current_period_start = now
      current_period_end = calculated
      amount_paid = amount
    
    Update profiles:
      plan = plan
    
    Insert payment_history row
    
    Return { success: true, subscription }

STEP 4 — Success UI:

After payment verified:

  Show success overlay (full screen):

    "Your account has been upgraded.
     All Pro features are now active."

    → Closes overlay, refreshes user plan

    "A receipt has been sent to

BACKEND ROUTES TO ADD
═══════════════════════════════════════

POST /api/payments/create-order
  Body: { userId, plan, billingCycle }
  → Creates Razorpay order
  → Returns { orderId, amount, keyId }

POST /api/payments/verify
  Body: { razorpay_order_id,
          razorpay_payment_id,
          razorpay_signature,
          userId, plan, billingCycle }
  → Verifies payment signature
  → Updates subscription + profile
  → Records payment history
  → Returns { success, subscription }

POST /api/payments/start-trial
  Body: { userId }
  → Validates trial not used
  → Creates trial subscription
  → Returns { success, trialEndsAt }

POST /api/payments/cancel
  Body: { userId }
  → Marks subscription cancelled
  → Returns { success, accessUntil }

GET /api/payments/subscription/:userId
  → Returns current subscription
  → Includes usage data

GET /api/payments/history/:userId
  → Returns payment history

POST /api/contact/college-inquiry
  Body: { institutionName, contactName,
          email, phone, studentCount,
          message }
  → Saves to college_inquiries table
  → Returns { success }

POST /api/usage/increment-ai
  Body: { userId }
  → Checks and resets daily counter
  → Increments ai_messages_today
  → Returns { allowed, remaining }

POST /api/projects/archive
  Body: { projectId, userId }
  → Sets status = 'archived'
  → Updates usage count
  → Returns { success }
